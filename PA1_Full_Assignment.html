6. Imputing Missing Values


# Count missing values
sum(is.na(data$steps))

# Create copy of dataset
data_imputed <- data

# Replace NA values with mean for that 5-min interval
for (i in 1:nrow(data_imputed)) {
  if (is.na(data_imputed$steps[i])) {
    interval_val <- data_imputed$interval[i]
    data_imputed$steps[i] <- avg_interval$steps[avg_interval$interval == interval_val]
  }
}

# Verify no missing values remain
sum(is.na(data_imputed$steps))

7. Histogram After Imputing Missing Values


total_steps_imputed <- aggregate(steps ~ date, data_imputed, sum)

hist(total_steps_imputed$steps,
     main="Histogram of Total Steps per Day (Imputed)",
     xlab="Total Steps per Day")

8. Panel Plot Comparing Weekdays and Weekends


data_imputed$weekday <- ifelse(weekdays(data_imputed$date) %in% 
                                c("Saturday", "Sunday"),

library(lattice)

avg_interval_type <- aggregate(steps ~ interval + weekday,

       type="l",
